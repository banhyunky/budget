이 앱은 GitHub Pages에 코드만 올라가 있어요. 실제 가계 데이터는 이 기기 또는 현기님이 만든 Firebase 프로젝트에만 저장되고, 공개 저장소에는 남지 않아요.

PIN은 둘이 같은 가계부를 보는지 확인하는 용도예요. 인터넷에 공개된 앱이므로, 실제 금액을 쓰기 전에 Firebase 콘솔 → Firestore 규칙에서 접근을 제한해 두는 걸 권해요.
Firestore 규칙 예시 보기 
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /households/{id} {
      // 테스트용: 누구나 접근 가능 (PIN은 앱에서 확인)
      allow read, write: if true;
      // 실제 사용 시: 아래처럼 본인 인증을 붙이는 걸 권장
      // allow read, write: if request.auth != null;
    }
  }
}
